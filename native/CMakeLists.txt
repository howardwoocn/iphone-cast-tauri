cmake_minimum_required(VERSION 3.16)
project(iphone_cast_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(NO_MARCH_NATIVE ON CACHE BOOL "Portable binary" FORCE)
set(NO_X11_DEPS ON CACHE BOOL "No X11 on Windows" FORCE)
add_compile_definitions(_WIN32_WINNT=0x0A00)
add_subdirectory(../vendor/UxPlay upstream EXCLUDE_FROM_ALL)
set_source_files_properties(../vendor/UxPlay/uxplay.cpp PROPERTIES COMPILE_DEFINITIONS "main=uxplay_main")
add_executable(cast-receiver receiver.cpp ../vendor/UxPlay/uxplay.cpp)
target_include_directories(cast-receiver PRIVATE ../vendor/UxPlay)
target_compile_definitions(cast-receiver PRIVATE IPHONE_CAST_BRIDGE)
target_link_libraries(cast-receiver PRIVATE renderers airplay user32 gdi32 shell32)
target_include_directories(renderers PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(airplay PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/third_party)
add_executable(geometry-tests geometry_tests.cpp)
enable_testing()
add_test(NAME geometry COMMAND geometry-tests)
